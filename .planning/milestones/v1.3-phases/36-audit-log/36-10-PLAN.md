<objective>
Turn the raw `{ field: { from, to } }` blob stored on an audit row into the labelled, typed, ordered
`AuditFieldChange[]` the renderer consumes — as a pure module, because the suite mocks the database
and this is the only way any of it can be tested.

Purpose: the UI-SPEC's renderer is dumb by design. Every database-dependent decision (resolving an
owner id to a person, reading a custom field's user-authored name) is made upstream, and every
locale-dependent decision (date, number and list formatting) is made in the client component. This
module is the seam between them.

Output: additive type declarations in `src/lib/timeline/types.ts`, plus
`src/lib/audit/present.ts` and its test.

**This plan does NOT add `'audit'` to `TimelineEntryKind` or to the `TimelineEntry` union.** Doing so
would fail `tsc` in `timeline-entry.tsx` immediately — which is Phase 35's exhaustive `never` gate
working as designed. The union edit lands in 36-13 together with the renderer branch that satisfies
it.
</objective>

export type AuditActorKind = 'user' | 'workflow_run' | 'api_key' | 'import' | 'system'
export type AuditAction = 'created' | 'updated' | 'deleted'

export type AuditValue =
  | { type: 'empty' }
  | { type: 'text'; value: string }
  | { type: 'number'; value: number }
  | { type: 'boolean'; value: boolean }
  | { type: 'date'; iso: string; withTime: boolean }
  | { type: 'list'; items: string[] }
  | { type: 'reference'; label: string | null }
  | { type: 'files'; count: number }
  | { type: 'json'; value: string }

export interface AuditFieldChange {
  field: string           // column name, or `custom:&lt;definitionId&gt;`
  label: string           // resolved and localized server-side
  from: AuditValue | null // null on a `created` entry
  to: AuditValue
}

export interface AuditTimelineEntry extends TimelineEntryBase {
  kind: 'audit'
  action: AuditAction
  entityType: EntityType
  actorKind: AuditActorKind
  actor: { id: string; name: string | null; email: string } | null
  workflowRun: { runId: string; workflowId: string; workflowName: string } | null
  apiKeyName: string | null
  changes: AuditFieldChange[]
}
