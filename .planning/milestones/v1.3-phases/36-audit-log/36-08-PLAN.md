export const AUDIT_RETENTION_KEY = "audit.retention_days"
export const RETENTION_MIN = 1
export const RETENTION_MAX = 3650

/** null ⇒ unset, unparseable or out of range. For the pruner, null means DELETE NOTHING. */
export async function readRetentionDays(): Promise&lt;number | null&gt;

/** Upsert on the key. Rejects out-of-range input before it reaches the database. */
export async function writeRetentionDays(days: number): Promise&lt;{ success: true } | { success: false; error: string }&gt;

export interface AuditStats { entryCount: number; oldestEntryAt: Date | null }
export async function readAuditStats(): Promise&lt;AuditStats&gt;
</interfaces>
