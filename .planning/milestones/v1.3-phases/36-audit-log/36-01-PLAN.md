  importSessionId?: string | null
}

export function getCurrentActor(): AuditActor | undefined
export function runWithActor&lt;T&gt;(actor: AuditActor, fn: () =&gt; T | Promise&lt;T&gt;): T | Promise&lt;T&gt;
